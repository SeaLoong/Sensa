// Optional adapter: imported only by the editor probe after dependency checks.
// Reads the user's installed SPS library; does not bundle or patch VRCFury.
Shader "Hidden/Sensa/SpsGpuProbe"
{
    Properties
    {
        _SPS_Configured ("Configured", Float) = 0
        _SPS_IdLow ("Id low", Float) = 0
        _SPS_IdHigh ("Id high", Float) = 0
        _SPS_PlayerIdLow ("Player low", Float) = 0
        _SPS_PlayerIdHigh ("Player high", Float) = 0
        _SPS_Overrun ("Overrun", Float) = 0
    }
    SubShader
    {
        Pass
        {
            Cull Off ZWrite Off ZTest Always Blend Off
            CGPROGRAM
            #pragma target 5.0
            #pragma vertex vert_img
            #pragma fragment frag
            #include "UnityCG.cginc"
            float4 _SensaSourceSize;
            // SPS hashes against the originating camera size, not this 36x1 readback target.
            #define _ScreenParams _SensaSourceSize
            #include "Packages/com.vrcfury.vrcfury/SPS/deform/sps_deform_curve.cginc"
            #undef _ScreenParams

            float4 frag(v2f_img input) : SV_Target
            {
                uint column = min((uint)(input.uv.x * 36), 35u);
                SpsTexture source = SPS_GET_TEX(_VFGridFinal);
                int slot = -1;
                bool found = _SPS_Configured > .5 && sps_try_find_cell(source,
                    sps_hash_id(sps_id(), sps_player_id()), sps_id(), sps_player_id(), SPS_PRODUCT_PLUG, slot);
                if (!found) return column == 0 ? float4(1947, 1, 0, 0) : 0;
                SpsCell cell = sps_get_cell(source, slot);
                if (cell.read_uint(SPS_HEADER_VERSION_INDEX) != SPS_VERSION_SPS)
                    return column == 0 ? float4(1947, 1, 0, 0) : 0;
                float worldLength = cell.read_float(sps_cell_pixel_index_from_payload_index(SPS_RESOLVER_METADATA_LENGTH_INDEX));
                float3 entryForward = sps_read_resolver_chain_forward(cell, 1);
                bool usable = isfinite(worldLength) && worldLength > .001 && worldLength < 10
                    && all(isfinite(entryForward)) && dot(entryForward, entryForward) > .25;
                if (column == 0) return float4(1947, 1, usable ? 1 : 0, worldLength);
                if (!usable) return 0;
                if (column == 1) return float4(sps_read_resolver_chain_world(cell, 1), 1);
                if (column == 2) return float4(normalize(entryForward), 1);
                float strength, radiusMultiplier;
                float3 position, forward, up;
                sps_deform_walk_chain(cell, worldLength, worldLength * (column - 3) / 32.0,
                    strength, radiusMultiplier, position, forward, up);
                // Unblended model centerline, not final skinned/deformed mesh vertices.
                return float4(position, strength);
            }
            ENDCG
        }
    }
    Fallback Off
}
